Q) Find Second Largest no of an array
---------------------------------
using System;

class Program
{
    static void Main()
    {
        int[] arr = { 10, 20, 5, 40, 30 };

        int largest = int.MinValue;
        int secondLargest = int.MinValue;

        foreach (int num in arr)
        {
            if (num > largest)
            {
                secondLargest = largest;
                largest = num;
            }
            else if (num > secondLargest && num != largest)
            {
                secondLargest = num;
            }
        }

        Console.WriteLine($"Largest = {largest}");
        Console.WriteLine($"Second Largest = {secondLargest}");
    }
}

<====Program to Find Vowels in a Name====>
==========================================
using System;

public class HelloWorld
{
    public static void Main(string[] args)
    {
        Console.WriteLine("Enter the Name");
        string name = Console.ReadLine();
        int count=0;
        
        Console.Write("Vowels are: ");
        
    
        foreach (char ch in name.ToLower())
        {
            if (ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u')
            {
                Console.Write(ch + " ");
                count ++;
            }
        }
         Console.WriteLine();
        Console.Write($"Number of vowels: {count}");
        
    }
}

Prime Number
using System;
public class Program
{
	public static void Main()
	{
		Console.WriteLine("Enter the Number");
		int num = Convert.ToInt32(Console.ReadLine());
		bool isPrime = true;
		if(num<=1)
		{
			 isPrime = false;
		}
		else{
			for(int i = 2; i<= num/2; i++)
			{
				if(num % i == 0)
				{
					isPrime = false;
					break;
				}
			}
		}
		if (isPrime)
            {
                Console.WriteLine($"{num} is a prime number.");
            }
            else
            {
                Console.WriteLine($"{num} is not a prime number.");
            }
	}
}

using System;
using System.Linq;	
public class Program
{
	public static void Main()
	{
		int[] arr = {1,3,4,2,4,6,3,1,3,2};
		
		var res = arr.Distinct();
		
		foreach( var item in res)
		{
			Console.Write(item + " " );
		}
	}
}
